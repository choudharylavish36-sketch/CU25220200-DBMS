Q1. Demonstrate the ABS() function.
SELECT ABS(-25) AS Positive_Value,
       ABS(-100) AS Positive_Number,
       ABS(45) AS Original_Positive_Value;
Q2. Demonstrate the ROUND() function.
SELECT ROUND(25.678, 2) AS Rounded_Value,
       ROUND(45.456, 1) AS Rounded_One_Digit,
       ROUND(78.567) AS Rounded_Integer;
Q3. Demonstrate CEIL() and FLOOR() functions.
SELECT CEIL(25.3) AS Ceiling_Value,
       FLOOR(25.8) AS Floor_Value,
       CEIL(10.1) AS Next_Integer,
       FLOOR(10.9) AS Previous_Integer;
Q4. Demonstrate the MOD() function.
SELECT MOD(17, 5) AS Remainder1,
       MOD(20, 4) AS Remainder2,
       MOD(15, 2) AS Remainder3;
Q5. Display employees' salaries rounded to the nearest integer.
SELECT Emp_ID,
       Emp_Name,
       Department,
       Salary,
       ROUND(Salary) AS Rounded_Salary
FROM Employee;
B. Aggregate Functions
Q6. Count the total number of students.
SELECT COUNT(*) AS Total_Students
FROM Student;

SELECT COUNT(*) AS Students_With_City
FROM Student
WHERE City IS NOT NULL;
Q7. Calculate the total salary of all employees.
SELECT SUM(Salary) AS Total_Salary
FROM Employee;

SELECT Department,
       SUM(Salary) AS Department_Total_Salary
FROM Employee
GROUP BY Department;
Q8. Calculate the average salary of employees.
SELECT AVG(Salary) AS Average_Salary
FROM Employee;

SELECT Department,
       AVG(Salary) AS Department_Average_Salary
FROM Employee
GROUP BY Department;
Q9. Find the maximum salary of employees.
SELECT MAX(Salary) AS Highest_Salary
FROM Employee;

SELECT Department,
       MAX(Salary) AS Department_Highest_Salary
FROM Employee
GROUP BY Department;
Q10. Find the minimum salary of employees.
SELECT MIN(Salary) AS Lowest_Salary
FROM Employee;

SELECT Department,
       MIN(Salary) AS Department_Lowest_Salary
FROM Employee
GROUP BY Department;
C. Character Functions
Q11. Display student names in uppercase.
SELECT Name AS Original_Name,
       UPPER(Name) AS Uppercase_Name
FROM Student;

SELECT UPPER(Name) AS Uppercase_Name
FROM Student
WHERE Name IS NOT NULL;
Q12. Display student names in lowercase.
SELECT Name AS Original_Name,
       LOWER(Name) AS Lowercase_Name
FROM Student;

SELECT LOWER(Name) AS Lowercase_Name
FROM Student
WHERE Name IS NOT NULL;
Q13. Find the length of student names.
SELECT Name,
       LENGTH(Name) AS Name_Length
FROM Student;

SELECT Name,
       LENGTH(Name) AS Name_Length
FROM Student
WHERE LENGTH(Name) > 5;
Q14. Demonstrate the CONCAT() function.
SELECT CONCAT('Lavish', ' ', 'Kumar') AS Full_Name;

SELECT CONCAT('Computer', ' ', 'Science') AS Course_Name;

SELECT CONCAT('Student Name: ', Name) AS Student_Details
FROM Student;
Q15. Demonstrate the SUBSTRING() function.
SELECT SUBSTRING('DATABASE', 1, 4) AS First_Four_Characters;

SELECT SUBSTRING('COMPUTER', 1, 3) AS First_Three_Characters;

SELECT SUBSTRING('PROGRAMMING', 4, 5) AS Selected_Characters;
D. Conversion Functions
Q16. Convert a number into a string.
SELECT CAST(12345 AS CHAR) AS String_Value;

SELECT CAST(9876 AS CHAR) AS Converted_Number;

SELECT CONCAT('Roll Number: ', CAST(101 AS CHAR))
       AS Student_Roll_Number;
Q17. Convert a string into a numeric value.
SELECT CAST('12345' AS UNSIGNED) AS Numeric_Value;

SELECT CAST('250' AS UNSIGNED) + 50 AS Total_Value;

SELECT CAST('125.75' AS DECIMAL(10,2)) + 10
       AS Calculated_Value;
Q18. Convert a date into a different format.
SELECT DATE_FORMAT('2026-10-14', '%d-%m-%Y')
       AS Date_Format1;

SELECT DATE_FORMAT('2026-10-14', '%M %d, %Y')
       AS Date_Format2;

SELECT DATE_FORMAT('2026-10-14', '%Y/%m/%d')
       AS Date_Format3;
Q19. Demonstrate the CAST() function.
SELECT CAST('123.45' AS DECIMAL(10,2))
       AS Decimal_Value;

SELECT CAST(500 AS CHAR)
       AS String_Value;

SELECT CAST('2026-10-14' AS DATE)
       AS Date_Value;
Q20. Demonstrate the CONVERT() function.
SELECT CONVERT('12345', UNSIGNED)
       AS Integer_Value;

SELECT CONVERT('123.45', DECIMAL(10,2))
       AS Decimal_Value;

SELECT CONVERT('2026-10-14', DATE)
       AS Converted_Date;
E. Date Functions
Q21. Display the current date.
SELECT CURDATE() AS Current_Date;

SELECT CURRENT_DATE() AS Today_Date;

SELECT NOW() AS Current_Date_And_Time;
Q22. Extract the year, month and day from a date.
SELECT YEAR('2026-10-14') AS Year_Value,
       MONTH('2026-10-14') AS Month_Value,
       DAY('2026-10-14') AS Day_Value;

SELECT MONTHNAME('2026-10-14') AS Month_Name,
       DAYNAME('2026-10-14') AS Day_Name;
Q23. Find the difference between two dates.
SELECT DATEDIFF('2026-10-14', '2026-10-01')
       AS Date_Difference;

SELECT DATEDIFF(CURDATE(), '2026-01-01')
       AS Days_From_January;

SELECT DATEDIFF('2026-12-31', '2026-01-01')
       AS Total_Days;
Q24. Add days to a date.
SELECT DATE_ADD('2026-10-14', INTERVAL 10 DAY)
       AS Date_After_10_Days;

SELECT DATE_ADD('2026-10-14', INTERVAL 1 MONTH)
       AS Date_After_One_Month;

SELECT DATE_ADD(CURDATE(), INTERVAL 7 DAY)
       AS Date_After_One_Week;
Q25. Display employees' joining dates in the required format.
ALTER TABLE Employee
ADD COLUMN Joining_Date DATE;

UPDATE Employee
SET Joining_Date = '2025-01-15'
WHERE Emp_ID = 1;

SELECT Emp_ID,
       Emp_Name,
       Joining_Date,
       DATE_FORMAT(Joining_Date, '%d-%m-%Y')
       AS Formatted_Joining_Date
FROM Employee;

SELECT Emp_Name,
       DATE_FORMAT(Joining_Date, '%M %d, %Y')
       AS Joining_Date_In_Words
FROM Employee
WHERE Joining_Date IS NOT NULL;
