Q5. Calculate the total amount as Price × Quantity.
CREATE TABLE Product (
    Product_ID INT PRIMARY KEY,
    Product_Name VARCHAR(50),
    Price DECIMAL(10,2),
    Quantity INT
);

INSERT INTO Product
VALUES
(1, 'Laptop', 50000, 2),
(2, 'Mouse', 500, 5),
(3, 'Keyboard', 1200, 3),
(4, 'Monitor', 10000, 2);

SELECT Product_ID,
       Product_Name,
       Price,
       Quantity,
       Price * Quantity AS Total_Amount
FROM Product;
B. Logical Operators
Q6. Demonstrate the AND operator using two conditions.
SELECT *
FROM Employee
WHERE Salary > 20000
  AND Department = 'IT';
Q7. Demonstrate the OR operator using two conditions.
SELECT *
FROM Employee
WHERE Department = 'IT'
   OR Department = 'HR';
Q8. Demonstrate the NOT operator using a condition.
SELECT *
FROM Employee
WHERE NOT Department = 'IT';
Q9. Find students belonging to a specific course and city.
SELECT *
FROM Student
WHERE Course = 'BCA'

Q10. Find employees belonging to either of two departments.
SELECT Emp_ID,
       Emp_Name,
       Department,
       Salary
FROM Employee
WHERE Department = 'IT'
   OR Department = 'HR';

Q25. Use arithmetic, logical, comparison and special operators together.
SELECT Product_ID,
       Product_Name,
       Price,
       Quantity,
       Price * Quantity AS Total_Amount
FROM Product
WHERE Price BETWEEN 100 AND 50000
  AND Quantity >= 2
  AND Product_Name LIKE 'L%'
  AND Price * Quantity > 1000;
