Q1. Demonstrate INNER JOIN to display matching records from two tables.
SELECT
    Student.Student_ID,
    Student.Name,
    Student.Course_ID,
    Course.Course_Name
FROM Student
INNER JOIN Course
    ON Student.Course_ID = Course.Course_ID;
Q2. Demonstrate LEFT JOIN to display all students and their course details.
SELECT
    Student.Student_ID,
    Student.Name,
    Student.Course_ID,
    Course.Course_Name
FROM Student
LEFT JOIN Course
    ON Student.Course_ID = Course.Course_ID;

Q4. Demonstrate CROSS JOIN between two tables.

Q5. Demonstrate SELF JOIN to display employees and their managers.
SELECT
    E.Emp_ID,
    E.Emp_Name AS Employee_Name,
    M.Emp_Name AS Manager_Name
FROM Employee E
LEFT JOIN Employee M
    ON E.Manager_ID = M.Emp_ID;

Q10. Display employees whose salary is greater than ₹30,000 with their department details.

    Employee.Emp_ID,
    Employee.Emp_Name,
    Employee.Salary,
    Department.Department_Name
FROM Employee
INNER JOIN Department
    ON Employee.Department = Department.Department_Name

Q23. Display employees who do not belong to any matching department.

    Employee.Emp_ID,

    ON Employee.Department = Department.Department_Name

Q24. Display departments that have no employees.
SELECT
    Department.Department_ID,
    Department.Department_Name
FROM Department
LEFT JOIN Employee
    ON Department.Department_Name = Employee.Department
WHERE Employee.Emp_ID IS NULL;
Q25. Display student names, course names and marks for students who scored more than 75.
SELECT
    Student.Student_ID,
    Student.Name,
    Course.Course_Name,
    Marks.Subject,
    Marks.Marks_Obtained
FROM Student
INNER JOIN Course
    ON Student.Course_ID = Course.Course_ID
INNER JOIN Marks
    ON Student.Student_ID = Marks.Student_ID
WHERE Marks.Marks_Obtained > 75
ORDER BY Marks.Marks_Obtained DESC;
